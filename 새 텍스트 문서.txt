    void TradingSystem::ResetAllTradersRandom(std::uint32_t seed)
    {
        std::mt19937 rng(seed);

        // 각 상인의 전문 분야 아이템 풀 정의 (기존 아이템 ID 0~14 기준)
        struct PoolEntry { int itemId; int minCount; int maxCount; int minStacks; int maxStacks; };
        const std::vector<PoolEntry> weaponPool = {
            { 11, 1, 1, 1, 2 },   // 소총
            { 12, 1, 1, 2, 4 },   // 소총 탄창
            { 13, 1, 1, 1, 2 },   // 권총
            { 14, 1, 1, 2, 4 },   // 권총 탄창
            { 2, 15, 30, 2, 5 }   // 공용 탄약
        };
        const std::vector<PoolEntry> medFoodPool = {
            { 0, 1, 5, 3, 6 },    // 붕대
            { 1, 1, 2, 3, 6 }     // 물
        };
        const std::vector<PoolEntry> gearPool = {
            { 3, 1, 1, 1, 2 },    // 상의
            { 4, 1, 1, 1, 2 },    // 하의
            { 7, 1, 1, 1, 2 },    // 조끼
            { 8, 1, 1, 1, 2 },    // 헬멧
            { 9, 1, 1, 1, 2 },    // 배낭
            { 10, 1, 1, 1, 2 }    // 군화
        };

        auto populateTrader = [&](TraderState& trader, const std::vector<PoolEntry>& pool)
        {
            trader.stashItems.clear();
            for (const auto& entry : pool)
            {
                std::uniform_int_distribution<int> stackDist(entry.minStacks, entry.maxStacks);
                std::uniform_int_distribution<int> countDist(entry.minCount, entry.maxCount);
                const int stacks = stackDist(rng);
                for (int s = 0; s < stacks; ++s)
                {
                    ItemStack item = MakeDefaultItemStack(entry.itemId, countDist(rng));
                    int freeX = 0, freeY = 0;
                    if (FindFirstFreeSlot(trader.stashItems, kTraderStashCols, kTraderStashRows,
                                          item.width, item.height, freeX, freeY))
                    {
                        trader.stashItems.push_back({ item, freeX, freeY });
                    }
                }
            }
        };

        populateTrader(traders_[0], weaponPool);
        populateTrader(traders_[1], medFoodPool);
        populateTrader(traders_[2], gearPool);
    }

    bool TradingSystem::StageFromTrader(TradeSession& session, int traderItemIndex, int targetStageX, int targetStageY)
    {
        auto& trader = GetTrader(session.activeTrader);
        if (traderItemIndex < 0 || traderItemIndex >= static_cast<int>(trader.stashItems.size()))
            return false;

        const GridItemPlacement& src = trader.stashItems[traderItemIndex];
        if (!CanPlaceInStage(session.buyOffer, targetStageX, targetStageY, src.stack.width, src.stack.height))
            return false;

        StagedItem staged;
        staged.stack = src.stack;
        staged.stageX = targetStageX;
        staged.stageY = targetStageY;
        staged.originX = src.gridX;
        staged.originY = src.gridY;

        session.buyOffer.push_back(staged);
        trader.stashItems.erase(trader.stashItems.begin() + traderItemIndex);
        return true;
    }

    bool TradingSystem::StageFromPlayer(TradeSession& session, Hideout& hideout, int playerItemIndex, int targetStageX, int targetStageY)
    {
        auto& playerStash = hideout.MutableMainStash(); // 10x8 일반 창고
        if (playerItemIndex < 0 || playerItemIndex >= static_cast<int>(playerStash.size()))
            return false;

        const GridItemPlacement& src = playerStash[playerItemIndex];

        // 내부에 물건이 들어 있는 조끼·배낭은 판매 칸 이동 차단
        if (HasContainedItems(hideout, src))
        {
            session.statusMessage = "내부에 물건이 들어 있는 조끼·배낭은 판매할 수 없습니다.";
            return false;
        }

        if (!CanPlaceInStage(session.sellOffer, targetStageX, targetStageY, src.stack.width, src.stack.height))
            return false;

        StagedItem staged;
        staged.stack = src.stack;
        staged.stageX = targetStageX;
        staged.stageY = targetStageY;
        staged.originX = src.gridX;
        staged.originY = src.gridY;

        session.sellOffer.push_back(staged);
        playerStash.erase(playerStash.begin() + playerItemIndex);
        session.statusMessage.clear();
        return true;
    }

    bool TradingSystem::UnstageBuyItem(TradeSession& session, int stagedIndex)
    {
        if (stagedIndex < 0 || stagedIndex >= static_cast<int>(session.buyOffer.size()))
            return false;

        auto& trader = GetTrader(session.activeTrader);
        const StagedItem item = session.buyOffer[stagedIndex];
        int rx = item.originX;
        int ry = item.originY;
        if (!CanPlaceInGrid(trader.stashItems, kTraderStashCols, kTraderStashRows, rx, ry, item.stack.width, item.stack.height))
        {
            if (!FindFirstFreeSlot(trader.stashItems, kTraderStashCols, kTraderStashRows, item.stack.width, item.stack.height, rx, ry))
                return false;
        }

        trader.stashItems.push_back({ item.stack, rx, ry });
        session.buyOffer.erase(session.buyOffer.begin() + stagedIndex);
        return true;
    }

    bool TradingSystem::UnstageSellItem(TradeSession& session, Hideout& hideout, int stagedIndex)
    {
        if (stagedIndex < 0 || stagedIndex >= static_cast<int>(session.sellOffer.size()))
            return false;

        auto& playerStash = hideout.MutableMainStash();
        const StagedItem item = session.sellOffer[stagedIndex];
        int rx = item.originX;
        int ry = item.originY;
        if (!CanPlaceInGrid(playerStash, 10, 8, rx, ry, item.stack.width, item.stack.height))
        {
            if (!FindFirstFreeSlot(playerStash, 10, 8, item.stack.width, item.stack.height, rx, ry))
                return false;
        }

        playerStash.push_back({ item.stack, rx, ry });
        session.sellOffer.erase(session.sellOffer.begin() + stagedIndex);
        return true;
    }

    void TradingSystem::RollbackSession(TradeSession& session, Hideout& hideout)
    {
        // 원래 창고 좌표(originX, originY)로 역순 복귀시켜 점유 충돌 방지
        while (!session.buyOffer.empty())
        {
            UnstageBuyItem(session, static_cast<int>(session.buyOffer.size()) - 1);
        }
        while (!session.sellOffer.empty())
        {
            UnstageSellItem(session, hideout, static_cast<int>(session.sellOffer.size()) - 1);
        }
    }

    bool TradingSystem::CommitTrade(TradeSession& session, Hideout& hideout)
    {
        if (session.buyOffer.empty() && session.sellOffer.empty())
            return false;

        const int buyTotal = CalculateBuyTotal(session);
        const int sellTotal = CalculateSellTotal(session);
        const int currentRubles = hideout.GetRubles();
        const int nextRubles = currentRubles - buyTotal + sellTotal;

        if (nextRubles < 0)
        {
            session.statusMessage = "보유 루블이 부족합니다.";
            return false;
        }

        // 1) 스냅샷 생성 (실패 시 원자적 복구 보장)
        auto& trader = GetTrader(session.activeTrader);
        const auto backupPlayerStash = hideout.MutableMainStash();
        const auto backupTraderStash = trader.stashItems;
        const auto backupBuyOffer = session.buyOffer;
        const auto backupSellOffer = session.sellOffer;

        auto simulatedPlayerStash = backupPlayerStash;
        auto simulatedTraderStash = backupTraderStash;

        // 2) 위칸(구매 물품) -> 내 창고(10x8) 수용 검증
        for (const auto& buyItem : session.buyOffer)
        {
            int freeX = 0, freeY = 0;
            if (!FindFirstFreeSlot(simulatedPlayerStash, 10, 8, buyItem.stack.width, buyItem.stack.height, freeX, freeY))
            {
                session.statusMessage = "내 창고에 공간이 부족합니다.";
                return false;
            }
            simulatedPlayerStash.push_back({ buyItem.stack, freeX, freeY });
        }

        // 3) 아래칸(판매 물품) -> 상인 창고(10x8) 수용 검증
        for (const auto& sellItem : session.sellOffer)
        {
            int freeX = 0, freeY = 0;
            if (!FindFirstFreeSlot(simulatedTraderStash, kTraderStashCols, kTraderStashRows,
                                   sellItem.stack.width, sellItem.stack.height, freeX, freeY))
            {
                session.statusMessage = "상인 창고에 공간이 부족합니다.";
                return false;
            }
            simulatedTraderStash.push_back({ sellItem.stack, freeX, freeY });
        }

        // 4) 메모리 상태 반영 후 디스크 원자적 저장
        hideout.MutableMainStash() = std::move(simulatedPlayerStash);
        trader.stashItems = std::move(simulatedTraderStash);
        hideout.SetRubles(nextRubles);
        session.buyOffer.clear();
        session.sellOffer.clear();

        if (!hideout.Save())
        {
            // 저장 실패 시 거래 전 원본으로 완전 롤백
            hideout.MutableMainStash() = backupPlayerStash;
            trader.stashItems = backupTraderStash;
            hideout.SetRubles(currentRubles);
            session.buyOffer = backupBuyOffer;
            session.sellOffer = backupSellOffer;
            session.statusMessage = "저장에 실패하여 거래가 취소되었습니다.";
            return false;
        }

        session.statusMessage = "거래가 완료되었습니다.";
        return true;
    }

    int TradingSystem::EvaluateStackPrice(const ItemStack& stack) const
    {
        // 현재 단계에서는 구매가와 판매가를 동일하게 계산
        const int unitPrice = GetItemBasePriceRubles(stack.itemId);
        return unitPrice * std::max(1, stack.count);
    }

    int TradingSystem::CalculateBuyTotal(const TradeSession& session) const
    {
        int total = 0;
        for (const auto& item : session.buyOffer)
            total += EvaluateStackPrice(item.stack);
        return total;
    }

    int TradingSystem::CalculateSellTotal(const TradeSession& session) const
    {
        int total = 0;
        for (const auto& item : session.sellOffer)
            total += EvaluateStackPrice(item.stack);
        return total;
    }

    bool TradingSystem::CanPlaceInGrid(const std::vector<GridItemPlacement>& grid, int cols, int rows,
                                       int x, int y, int w, int h) const
    {
        if (x < 0 || y < 0 || x + w > cols || y + h > rows)
            return false;

        for (const auto& existing : grid)
        {
            const bool overlapX = (x < existing.gridX + existing.stack.width) && (x + w > existing.gridX);
            const bool overlapY = (y < existing.gridY + existing.stack.height) && (y + h > existing.gridY);
            if (overlapX && overlapY)
                return false;
        }
        return true;
    }

    bool TradingSystem::CanPlaceInStage(const std::vector<StagedItem>& staged, int x, int y, int w, int h) const
    {
        if (x < 0 || y < 0 || x + w > kTradeStageCols || y + h > kTradeStageRows)
            return false;

        for (const auto& existing : staged)
        {
            const bool overlapX = (x < existing.stageX + existing.stack.width) && (x + w > existing.stageX);
            const bool overlapY = (y < existing.stageY + existing.stack.height) && (y + h > existing.stageY);
            if (overlapX && overlapY)
                return false;
        }
        return true;
    }

    bool TradingSystem::FindFirstFreeSlot(const std::vector<GridItemPlacement>& grid, int cols, int rows,
                                          int w, int h, int& outX, int& outY) const
    {
        for (int y = 0; y <= rows - h; ++y)
        {
            for (int x = 0; x <= cols - w; ++x)
            {
                if (CanPlaceInGrid(grid, cols, rows, x, y, w, h))
                {
                    outX = x;
                    outY = y;
                    return true;
                }
            }
        }
        return false;
    }

    bool TradingSystem::HasContainedItems(const Hideout& hideout, const GridItemPlacement& placement) const
    {
        // 조끼/배낭 아이템 자체에 저장된 내부 수납물 유무 확인
        return placement.stack.hasNestedItems || hideout.DoesContainerStackHaveItems(placement.stack.stackId);
    }

    QuestProgress* TradingSystem::FindQuestProgress(int questId)
    {
        for (auto& qp : questProgress_)
        {
            if (qp.questId == questId)
                return &qp;
        }
        return nullptr;
    }

    bool TradingSystem::AcceptQuest(int questId, Hideout& hideout)
    {
        QuestProgress* qp = FindQuestProgress(questId);
        if (!qp || qp->status != QuestStatus::Available)
            return false;

        qp->status = QuestStatus::Active;
        qp->currentCount = 0;
        if (!hideout.Save())
        {
            qp->status = QuestStatus::Available;
            return false;
        }
        return true;
    }

    bool TradingSystem::DeliverQuestItems(int questId, Hideout& hideout)
    {
        const QuestDef* def = nullptr;
        for (const auto& d : questDefs_)
            if (d.questId == questId) { def = &d; break; }

        QuestProgress* qp = FindQuestProgress(questId);
        if (!def || !qp || qp->status != QuestStatus::Active)
            return false;
        if (def->objectiveType != QuestObjectiveType::DeliverItem)
            return false;

        const int needed = def->requiredCount - qp->currentCount;
        if (needed <= 0)
            return false;

        const auto backupStash = hideout.MutableMainStash();
        const int backupCount = qp->currentCount;
        const QuestStatus backupStatus = qp->status;

        const int consumed = hideout.ConsumeFromMainStash(def->targetItemId, needed);
        if (consumed <= 0)
            return false;

        qp->currentCount += consumed;
        if (qp->currentCount >= def->requiredCount)
        {
            qp->currentCount = def->requiredCount;
            qp->status = QuestStatus::ReadyToComplete;
        }

        if (!hideout.Save())
        {
            hideout.MutableMainStash() = backupStash;
            qp->currentCount = backupCount;
            qp->status = backupStatus;
            return false;
        }
        return true;
    }

    bool TradingSystem::CompleteQuest(int questId, Hideout& hideout)
    {
        const QuestDef* def = nullptr;
        for (const auto& d : questDefs_)
            if (d.questId == questId) { def = &d; break; }

        QuestProgress* qp = FindQuestProgress(questId);
        if (!def || !qp || qp->status != QuestStatus::ReadyToComplete)
            return false;

        const auto backupStash = hideout.MutableMainStash();
        const int backupRubles = hideout.GetRubles();
        auto& trader = GetTrader(def->trader);
        const int backupLoyalty = trader.loyaltyLevel;

        if (def->rewardItemId >= 0 && def->rewardItemCount > 0)
        {
            ItemStack rewardStack = MakeDefaultItemStack(def->rewardItemId, def->rewardItemCount);
            int freeX = 0, freeY = 0;
            if (!FindFirstFreeSlot(hideout.MutableMainStash(), 10, 8, rewardStack.width, rewardStack.height, freeX, freeY))
                return false; // 창고 공간 부족 시 보상 증발 방지
            hideout.MutableMainStash().push_back({ rewardStack, freeX, freeY });
        }

        hideout.SetRubles(backupRubles + def->rewardRubles);
        trader.loyaltyLevel += def->rewardLoyalty;
        qp->status = QuestStatus::Completed;

        if (!hideout.Save())
        {
            hideout.MutableMainStash() = backupStash;
            hideout.SetRubles(backupRubles);
            trader.loyaltyLevel = backupLoyalty;
            qp->status = QuestStatus::ReadyToComplete;
            return false;
        }
        return true;
    }

    void TradingSystem::OnRaidEnemyKilled()
    {
        for (const auto& def : questDefs_)
        {
            if (def.objectiveType != QuestObjectiveType::KillEnemies)
                continue;
            QuestProgress* qp = FindQuestProgress(def.questId);
            if (qp && qp->status == QuestStatus::Active)
            {
                qp->currentCount = std::min(def.requiredCount, qp->currentCount + 1);
                if (qp->currentCount >= def.requiredCount)
                    qp->status = QuestStatus::ReadyToComplete;
            }
        }
    }

    void TradingSystem::OnRaidPositionUpdated(float x, float z)
    {
        for (const auto& def : questDefs_)
        {
            if (def.objectiveType != QuestObjectiveType::VisitLocation)
                continue;
            QuestProgress* qp = FindQuestProgress(def.questId);
            if (qp && qp->status == QuestStatus::Active)
            {
                const float dx = x - def.targetX;
                const float dz = z - def.targetZ;
                if ((dx * dx + dz * dz) <= (def.targetRadius * def.targetRadius))
                {
                    qp->currentCount = def.requiredCount;
                    qp->status = QuestStatus::ReadyToComplete;
                }
            }
        }
    }

    void TradingSystem::OnRaidExtracted()
    {
        for (const auto& def : questDefs_)
        {
            if (def.objectiveType != QuestObjectiveType::ExtractSuccess)
                continue;
            QuestProgress* qp = FindQuestProgress(def.questId);
            if (qp && qp->status == QuestStatus::Active)
            {
                qp->currentCount = def.requiredCount;
                qp->status = QuestStatus::ReadyToComplete;
            }
        }
    }

        void DrawGridCells(ImDrawList* drawList, ImVec2 origin, int cols, int rows, float cellSize)
        {
            const ImU32 bgCol = IM_COL32(28, 32, 30, 210);
            const ImU32 lineCol = IM_COL32(75, 85, 80, 220);
            drawList->AddRectFilled(origin, ImVec2(origin.x + cols * cellSize, origin.y + rows * cellSize), bgCol);
            for (int r = 0; r <= rows; ++r)
            {
                const float y = origin.y + r * cellSize;
                drawList->AddLine(ImVec2(origin.x, y), ImVec2(origin.x + cols * cellSize, y), lineCol);
            }
            for (int c = 0; c <= cols; ++c)
            {
                const float x = origin.x + c * cellSize;
                drawList->AddLine(ImVec2(x, origin.y), ImVec2(x, origin.y + rows * cellSize), lineCol);
            }
        }

    void TradingHud::Render(Trading::TradingSystem& trading, Trading::TradeSession& session, Hideout& hideout)
    {
        if (!session.isOpen)
            return;

        // Esc 또는 F 입력 시 미확정 물품 원위치 롤백 후 창 닫기
        if (ImGui::IsKeyPressed(ImGuiKey_Escape) || ImGui::IsKeyPressed(ImGuiKey_F))
        {
            trading.CloseSession(session, hideout);
            return;
        }

        const ImGuiViewport* viewport = ImGui::GetMainViewport();
        ImGui::SetNextWindowPos(viewport->WorkPos);
        ImGui::SetNextWindowSize(viewport->WorkSize);

        const ImGuiWindowFlags flags = ImGuiWindowFlags_NoDecoration | ImGuiWindowFlags_NoMove | ImGuiWindowFlags_NoSavedSettings;
        ImGui::PushStyleColor(ImGuiCol_WindowBg, ImVec4(0.08f, 0.10f, 0.09f, 0.92f));

        if (ImGui::Begin("##TradingMainWindow", nullptr, flags))
        {
            RenderTopHeader(trading, session, hideout);
            ImGui::Separator();

            const ImVec2 avail = ImGui::GetContentRegionAvail();
            const float spacing = ImGui::GetStyle().ItemSpacing.x;
            const float leftWidth = avail.x * 0.36f;
            const float centerWidth = avail.x * 0.28f;
            const float rightWidth = avail.x - leftWidth - centerWidth - spacing * 2.0f;

            ImGui::SameLine();
            RenderCenterDealPanel(trading, session, hideout, centerWidth, avail.y);
            ImGui::SameLine();

            if (session.activeTab == Trading::TradingSubTab::Trade)
                RenderTraderRightPanel(trading, session, hideout, rightWidth, avail.y);
            else
                RenderTraderQuestPanel(trading, session, hideout, rightWidth, avail.y);
        }
        ImGui::End();
        ImGui::PopStyleColor();
    }

C++// Hideout.cpp — Depart() 내부 분기
bool Hideout::Depart(PlayableCharacterId selectedChar, std::string& outError)
{
    const int idx = static_cast<int>(selectedChar);
    if (idx < 0 || idx >= kMaxPlayableCharacters || !characters_[idx].unlocked)
    {
        outError = "선택한 캐릭터를 사용할 수 없습니다.";
        return false;
    }

    const bool isScav = (selectedChar != PlayableCharacterId::PmcMain);

    // 1. SCAV 출격 잠금 검사
    if (isScav && characters_[idx].deployLocked)
    {
        outError = "이 SCAV는 대기 중입니다. PMC로 한 번 출격해야 다시 출격할 수 있습니다.";
        return false;
    }

    // 2. 기존 생존 상태 및 SCAV 개인 장비 자동 보관 처리 (복사본 시뮬레이션)
    // ... (기존 Hideout::Depart의 창고 보관 및 무작위 장비 구성 로직 유지) ...

    // 3. 출격 유형에 따른 SCAV 잠금 상태 갱신
    auto backupCharacters = characters_;
    if (isScav)
    {
        // 해당 SCAV만 출격 잠금
        characters_[idx].deployLocked = true;
    }
    else
    {
        // PMC 출격 시 결과에 상관없이 모든 SCAV 출격 잠금 해제
        for (int s = 1; s < kMaxPlayableCharacters; ++s)
        {
            characters_[s].deployLocked = false;
        }
    }

    if (!Save())
    {
        characters_ = backupCharacters; // 저장 실패 시 롤백
        outError = "출격 상태 저장에 실패했습니다.";
        return false;
    }

    return true;
}

C++// HideoutHud.cpp — NPC F키 상호작용 시 열리는 대화 UI
void HideoutHud::RenderCharacterDialogue(Hideout& hideout, PlayableCharacterId targetChar, bool& isOpen)
{
    if (!isOpen) return;

    const auto& charState = hideout.GetCharacterState(targetChar);
    auto* mission = hideout.FindActiveOrNextPersonalMission(targetChar); // 미션 정의 + 진행도 조회

    ImGui::SetNextWindowSize(ImVec2(520.0f, 320.0f), ImGuiCond_FirstUseEver);
    ImGui::SetNextWindowPos(ImGui::GetMainViewport()->GetCenter(), ImGuiCond_FirstUseEver, ImVec2(0.5f, 0.5f));

    if (ImGui::Begin(charState.displayName.c_str(), &isOpen, ImGuiWindowFlags_NoCollapse))
    {
        // 1. 캐릭터 대사 영역
        if (mission)
        {
            ImGui::TextColored(ImVec4(0.85f, 0.90f, 0.75f, 1.0f), "[개인 미션: %s]", mission->def.title.c_str());
            ImGui::Spacing();
            if (mission->progress.status == Tarkov::Trading::QuestStatus::Available)
                ImGui::TextWrapped("\"%s\"", mission->def.introDialogue.c_str());
            else if (mission->progress.status == Tarkov::Trading::QuestStatus::Active)
                ImGui::TextWrapped("\"%s\"", mission->def.activeDialogue.c_str());
            else if (mission->progress.status == Tarkov::Trading::QuestStatus::ReadyToComplete)
                ImGui::TextWrapped("\"%s\"", mission->def.completeDialogue.c_str());
            else
                ImGui::TextWrapped("\"고마워. 덕분에 한숨 돌렸어.\"");

            ImGui::Spacing();
            ImGui::Text("진행도: %d / %d", mission->progress.currentCount, mission->def.requiredCount);
        }
        else
        {
            ImGui::TextWrapped("\"필요한 게 있으면 언제든 말해.\"");
        }

        ImGui::Separator();
        ImGui::Spacing();

        // 2. 개인 미션 수락 / 납품 / 보상 수령 버튼
        if (mission)
        {
            using Tarkov::Trading::QuestStatus;
            if (mission->progress.status == QuestStatus::Available)
            {
                if (ImGui::Button("개인 미션 수락", ImVec2(160.0f, 32.0f)))
                    hideout.AcceptPersonalMission(mission->def.missionId);
            }
            else if (mission->progress.status == QuestStatus::Active &&
                     mission->def.objectiveType == Tarkov::Trading::QuestObjectiveType::DeliverItem)
            {
                if (ImGui::Button("미션 물품 건네주기", ImVec2(160.0f, 32.0f)))
                    hideout.DeliverPersonalMissionItem(mission->def.missionId);
            }
            else if (mission->progress.status == QuestStatus::ReadyToComplete)
            {
                if (ImGui::Button("미션 완료 및 보상 받기", ImVec2(180.0f, 32.0f)))
                    hideout.CompletePersonalMission(mission->def.missionId);
            }
        }

        ImGui::Spacing();

        // 3. 캐릭터 전환 버튼 (SCAV 출격 잠금 상태 안내 포함)
        if (charState.deployLocked)
        {
            ImGui::TextColored(ImVec4(0.95f, 0.45f, 0.40f, 1.0f),
                "상태: 출격 잠금 (PMC로 출격하면 해제됩니다)");
        }

        if (ImGui::Button("이 캐릭터로 조작 전환", ImVec2(180.0f, 32.0f)))
        {
            hideout.SwitchActiveCharacter(targetChar);
            isOpen = false;
        }
        ImGui::SameLine();
        if (ImGui::Button("대화 종료 (Esc/F)", ImVec2(140.0f, 32.0f)))
        {
            isOpen = false;
        }
    }

C++// RaidScene.cpp — 사망 판정 처리 변경
void RaidScene::OnPlayerDied()
{
    if (raidFinished_)
        return;

    raidFinished_ = true;

    // 1. 프로필 정산: 휴대품/출전 장비 제거, 창고 보존, 상태 저장 (기존 Dead 정산 규칙 유지)
    RaidResultData result = BuildRaidResult(RaidOutcome::Dead);
    hideout_.ApplyRaidFailure(result); // 사망 상태 저장 (은신처 복귀 시 치료/복구 흐름 연결)

    // 2. 기존 별도 실패 HUD 대신 통합 탈출 결과 장면(ResultHud)으로 전환
    OpenRaidResultScene(result);
}
C++// ResultHud.cpp — 결과 상단 헤더 및 패널 분기
void ResultHud::Render(const RaidResultData& result, Hideout& hideout)
{
    // ... (기존 결과 화면 윈도우 설정) ...

    const bool isSuccess = (result.outcome == RaidOutcome::Extracted);

    if (isSuccess)
    {
        ImGui::TextColored(ImVec4(0.40f, 0.90f, 0.55f, 1.0f), "탈출 성공");
    }
    else
    {
        // 사망(Dead) 또는 MIA 시 탈출 UI에 "탈출실패" 표시
        ImGui::TextColored(ImVec4(0.95f, 0.30f, 0.30f, 1.0f), "탈출실패");
    }

    ImGui::Separator();
    ImGui::Text("생존/교전 시간: %02d:%02d", result.elapsedSeconds / 60, result.elapsedSeconds % 60);

    // 처치 기록 목록은 성공/실패 모두 표시
    RenderKillLogTable(result.kills);

    if (isSuccess)
    {
        // 기존 회수 장비 캐릭터 미리보기 및 새로 획득한 장비 목록 표시
        RenderExtractedGearPreview(result);
    }
    else
    {
        // 사망 시 손실된 출전 장비 목록 안내 (창고 보관품은 유지됨)
        ImGui::TextColored(ImVec4(0.85f, 0.65f, 0.65f, 1.0f), "출전 장비 및 레이드 획득품을 모두 잃었습니다.");
        RenderLostGearList(result.lostItems);
    }

    if (ImGui::Button("은신처로 복귀", ImVec2(200.0f, 40.0f)))
    {
        ReturnToHideout();
    }
}
